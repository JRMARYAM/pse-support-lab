   123			metrics.Submissions.WithLabelValues("failed").Inc()

   126		}

   129	
   130		var reply aspReply

   144			}

   135	func (l *limiter) allow(sender string, now time.Time) (bool, time.Duration) {
   136		l.mu.Lock()
   137		defer l.mu.Unlock()
   138		b, ok := l.buckets[sender]
   139		if !ok {
   140			b = &bucket{tokens: senderBurst, last: now}
   141			l.buckets[sender] = b
   142		}
   143		b.tokens = math.Min(senderBurst, b.tokens+now.Sub(b.last).Seconds()*senderRatePerSec)
   144		b.last = now
   145		if b.tokens >= 1 {
   146			b.tokens--
   147			return true, 0
   148		}
   149		wait := time.Duration((1 - b.tokens) / senderRatePerSec * float64(time.Second))
   150		return false, wait
